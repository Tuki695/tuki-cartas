  <h1>🎯 Adiviná tu carta - Tuki Burger 🎯</h1>
  <p>Elegí una carta y luego tratá de adivinarla tras mezclarse.</p>

  <div class="cartas" id="cartas"></div>

  <p id="mensaje"></p>
  <button id="reiniciar" style="display:none;">Jugar de nuevo</button>

  <script>
    const valores = ['♥', '♠', '♦'];
    let cartaElegida = null;
    let estado = 'elegir'; // 'elegir' o 'adivinar'
    const contenedor = document.getElementById('cartas');
    const mensaje = document.getElementById('mensaje');
    const btn = document.getElementById('reiniciar');

    function mostrarCartas() {
      contenedor.innerHTML = '';
      valores.forEach((v, i) => {
        const div = document.createElement('div');
        div.classList.add('carta');
        div.dataset.valor = v;
        div.onclick = () => {
          if (estado === 'elegir') {
            cartaElegida = v;
            mensaje.textContent = 'Se mezclaron las cartas... ¿Dónde está tu carta?';
            mezclar();
            estado = 'adivinar';
          } else if (estado === 'adivinar') {
            if (v === cartaElegida) {
              mensaje.textContent = '🎉 ¡Ganaste una Coca o 20% off!';
            } else {
              mensaje.textContent = '😅 No era esa, probá otra vez.';
            }
            btn.style.display = 'inline';
          }
        };
        contenedor.appendChild(div);
      });
    }

    function mezclar() {
      valores.sort(() => Math.random() - 0.5);
      mostrarCartas();
    }

    btn.onclick = () => {
      estado = 'elegir';
      cartaElegida = null;
      mensaje.textContent = '';
      btn.style.display = 'none';
      mostrarCartas();
    };

    mostrarCartas();
  </script>
